cmake_minimum_required(VERSION 3.16)
project(bpl_interpreter LANGUAGES CXX)
enable_testing()
add_executable(bpl-interpreter src/prog3.cpp src/lex.cpp src/parserInterp.cpp src/val.cpp)
target_compile_features(bpl-interpreter PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(bpl-interpreter PRIVATE /W3 /EHsc)
else()
  target_compile_options(bpl-interpreter PRIVATE -Wall -Wextra)
endif()
add_executable(bpl-value-tests tests/value_operations_test.cpp src/val.cpp)
target_include_directories(bpl-value-tests PRIVATE src)
target_compile_features(bpl-value-tests PRIVATE cxx_std_17)
add_test(NAME value-operations COMMAND bpl-value-tests)
